pipeline {
    agent any

    stages {

        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Build Docker Image') {
            steps {
                sh 'docker build -t simple-ecommerce .'
            }
        }

        stage('Run Docker Container') {
            steps {
                sh 'docker rm -f ecommerce || true'
                sh 'docker run -d -p 8080:80 --name ecommerce simple-ecommerce'
            }
        }
    }
}
